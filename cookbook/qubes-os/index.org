** Open xterm (or terminal) in Qube as root on any QUBE
Ref: https://www.qubes-os.org/doc/vm-sudo/

#+begin_src bash
  # In dom0
  qvm-run --user root QUBENAME xterm
  # or
  qvm-run -u root QUBENAME qubes-run-terminal
  # or
  qvm-console-dispvm QUBENAME
#+end_src

** Access Qubes Global Clipboard using X Selection/Paste (XS_PRIMARY)
This is to be able to cut and paste from things like xterm.

- Send selected text (XA_PRIMARY) to XA_CLIPBOARD to Global Clipboard:  
#+begin_src bash
  # In same QUBE as window with highlighted text
  xclip -o | xclip -sel clip
  # then GLOBAL Copy (<CTRL>-<SHIFT>-<C>)
#+end_src

- Send Global Clipboard to XA_CLIPBOARD to XA_PRIMARY
#+begin_src bash
  # In same QUBE as window to be pasted
  # First GLOBAL Paste (<CTRL>-<SHIFT>-<P>) then
  xclip -o -sel clip | xlip -o
#+end_src

** Use split-gpg for LUKS dm-crypt devices
Ref: https://cryptsetup-team.pages.debian.net/cryptsetup/README.gnupg.html

- Create keyfile 


**** decrypt_qubes-gpg-client-wrapper
Minor modification to debian's [[https://salsa.debian.org/cryptsetup-team/cryptsetup/-/blob/debian/latest/debian/scripts/decrypt_gnupg-sc][cryptsetup | decrypt_gnupg-sc]] script
  - use ~~~qubes-gpg-client-wrapper~~~ instead of ~~~/usr/bin/gpg~~~
  -
    
#+begin_src sh
  #!/bin/sh

  local GPGEXECUTABLE=${GPGEXECUTABLE:-/usr/bin/qubes-gpg-client-wrapper}
  # Test
  echo $GPGEXECUTABLE
  exit

  if [ -d "/cryptroot/gnupghome" ]; then
      export GNUPGHOME="/cryptroot/gnupghome"
  fi

  run_gpg() {
      gpg --no-options --trust-model=always "$@"
  }
  decrypt_gpg () {
      local console _
      if ! GPG_TTY="$(tty)"; then
	  read console _ </proc/consoles
	  GPG_TTY="/dev/$console"
      fi
      export GPG_TTY

      if ! run_gpg --decrypt -- "$1"; then
	  return 1
      fi
      return 0
  }

  # `gpg-connect-agent LEARN /bye` is another (lighter) way, but it's
  # harder to retrieve the return code
  if ! run_gpg --batch --quiet --no-tty --card-status >/dev/null; then
      echo "Please insert OpenPGP SmartCard..." >&2
      until run_gpg --batch --quiet --no-tty --card-status; do
	  sleep 1
      done >/dev/null 2>&1
  fi

  if [ ! -x /usr/bin/gpg ]; then
      echo "$0: /usr/bin/gpg is not available" >&2
      exit 1
  fi

  if [ -z "$1" ] || [ ! -f "$1" ]; then
      echo "$0: missing key as argument" >&2
      exit 1
  fi

  decrypt_gpg "$1"
  exit $?
#+end_src

*** Share raw disk image to qubes
Ref: baiko's answer at
     https://forum.qubes-os.org/t/how-to-create-and-use-additional-volumes-for-vms/23959/4

- Create Raw disk image in storage qube
  
  #+begin_src bash
    # As root on storage Qube
    mkdir -p /usr/local/var/disks/
    dd if=/dev/urandom of='/usr/local/var/disks/DISKPURPOSE.raw' bs=1024M count=5
    # Where # is an integer between 0-63 and not used on same qube for other purpose
    losetup --show '/dev/loop#' '/usr/local/var/disks/DISKPURPOSE.raw'
  #+end_src
