   -  Download latest "Portable Emscripten SDK for Linux and OS X"
   -  Expand, update, activate and configure environment

      #+BEGIN_SRC sh
          tar -xzvf Downloads/emsdk-portable.tar.gz 
          cd emsdk_portable/
          ./emsdk update
          ./emsdk install latest
          ./emsdk activate latest
          source ./emsdk_env.sh 
      #+END_SRC
